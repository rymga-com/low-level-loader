cmake_minimum_required(VERSION 3.20)
project(rymga_loader_native C)

if (CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options($<$<CONFIG:Release>:-fstack-protector-strong>)
endif()
if (CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options($<$<CONFIG:Release>:-D_FORTIFY_SOURCE=3>)
    add_link_options($<$<CONFIG:Release>:-Wl,-z,relro,-z,now>)
endif()

option(RYMGA_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(RYMGA_ENABLE_FUZZER "Build the Clang libFuzzer target for the manifest parser" OFF)
option(RYMGA_EMBEDDED_CONFIG "Compile product settings for the Paper bootstrap" OFF)
option(RYMGA_SELF_CONTAINED "Require third-party runtime dependencies to be linked statically" OFF)
set(RYMGA_EMBEDDED_ENDPOINT "" CACHE STRING "Paper gateway HTTPS origin")
set(RYMGA_EMBEDDED_PRODUCT "" CACHE STRING "Paper product ID")
set(RYMGA_EMBEDDED_CHANNEL "stable" CACHE STRING "Paper release channel")
set(RYMGA_EMBEDDED_LOADER_VERSION "1.0.0" CACHE STRING "Paper loader version")
set(RYMGA_EMBEDDED_INSTALLATION_ID "paper" CACHE STRING "Paper installation ID")
set(RYMGA_EMBEDDED_KEY_ID "" CACHE STRING "Paper trusted Ed25519 key ID")
set(RYMGA_EMBEDDED_PUBLIC_KEY_HEX "" CACHE STRING "Paper trusted Ed25519 public key as 64 hex characters")
if (RYMGA_EMBEDDED_CONFIG)
    foreach(value RYMGA_EMBEDDED_ENDPOINT RYMGA_EMBEDDED_PRODUCT RYMGA_EMBEDDED_CHANNEL
            RYMGA_EMBEDDED_LOADER_VERSION RYMGA_EMBEDDED_INSTALLATION_ID RYMGA_EMBEDDED_KEY_ID)
        if ("${${value}}" STREQUAL "" OR "${${value}}" MATCHES "[\"\\\\]")
            message(FATAL_ERROR "${value} must be non-empty and contain no quote or backslash")
        endif()
    endforeach()
    if (NOT RYMGA_EMBEDDED_ENDPOINT MATCHES "^https://.+[^/]$")
        message(FATAL_ERROR "RYMGA_EMBEDDED_ENDPOINT must be an HTTPS origin without a trailing slash")
    endif()
    string(LENGTH "${RYMGA_EMBEDDED_PUBLIC_KEY_HEX}" RYMGA_EMBEDDED_PUBLIC_KEY_LENGTH)
    if (NOT RYMGA_EMBEDDED_PUBLIC_KEY_LENGTH EQUAL 64 OR NOT RYMGA_EMBEDDED_PUBLIC_KEY_HEX MATCHES "^[0-9A-Fa-f]+$")
        message(FATAL_ERROR "RYMGA_EMBEDDED_PUBLIC_KEY_HEX must contain 64 hexadecimal characters")
    endif()
endif()
configure_file(jni/rymga_embedded_config.h.in generated/rymga_embedded_config.h @ONLY)
if (RYMGA_ENABLE_FUZZER)
    if (NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
        message(FATAL_ERROR "RYMGA_ENABLE_FUZZER requires Clang")
    endif()
    add_link_options(-fsanitize=fuzzer-no-link,address,undefined)
endif()
if (RYMGA_ENABLE_SANITIZERS AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()

find_package(CURL REQUIRED)
set(RYMGA_JNI_INCLUDE_DIRS "" CACHE STRING "Optional JNI include directories for cross-compilation")
if (RYMGA_JNI_INCLUDE_DIRS)
    find_file(RYMGA_JNI_HEADER jni.h PATHS ${RYMGA_JNI_INCLUDE_DIRS} NO_DEFAULT_PATH REQUIRED)
    find_file(RYMGA_JNI_PLATFORM_HEADER jni_md.h PATHS ${RYMGA_JNI_INCLUDE_DIRS} NO_DEFAULT_PATH REQUIRED)
    set(JNI_INCLUDE_DIRS ${RYMGA_JNI_INCLUDE_DIRS})
else()
    find_package(JNI REQUIRED)
endif()
find_path(SODIUM_INCLUDE_DIR sodium.h REQUIRED)
find_library(SODIUM_LIBRARY sodium REQUIRED)

if (RYMGA_SELF_CONTAINED)
    if (NOT DEFINED VCPKG_TARGET_TRIPLET)
        message(FATAL_ERROR "RYMGA_SELF_CONTAINED requires a vcpkg static triplet")
    endif()
    if (WIN32 AND NOT VCPKG_TARGET_TRIPLET MATCHES "-static")
        message(FATAL_ERROR "Windows self-contained builds require a *-static triplet")
    endif()
    if (SODIUM_LIBRARY MATCHES "\\.(so|dylib|dll)(\\.|$)")
        message(FATAL_ERROR "RYMGA_SELF_CONTAINED resolved a shared libsodium library")
    endif()
    get_target_property(RYMGA_CURL_LIBRARY_TYPE CURL::libcurl TYPE)
    if (NOT RYMGA_CURL_LIBRARY_TYPE STREQUAL "STATIC_LIBRARY")
        message(FATAL_ERROR "RYMGA_SELF_CONTAINED requires static libcurl")
    endif()
    set(RYMGA_THIRD_PARTY_LICENSE_DIR "${CMAKE_CURRENT_BINARY_DIR}/third-party-licenses")
    file(MAKE_DIRECTORY "${RYMGA_THIRD_PARTY_LICENSE_DIR}")
    set(RYMGA_BUNDLED_DEPENDENCIES curl libsodium zlib)
    if (NOT VCPKG_TARGET_TRIPLET MATCHES "(windows|mingw)")
        list(APPEND RYMGA_BUNDLED_DEPENDENCIES openssl)
    endif()
    foreach(dependency IN LISTS RYMGA_BUNDLED_DEPENDENCIES)
        set(copyright "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/${dependency}/copyright")
        if (NOT EXISTS "${copyright}")
            message(FATAL_ERROR "Missing vcpkg copyright file for ${dependency}")
        endif()
        configure_file("${copyright}" "${RYMGA_THIRD_PARTY_LICENSE_DIR}/${dependency}.txt" COPYONLY)
    endforeach()
endif()

set(RYMGA_PLATFORM_SOURCE src/rymga_platform_posix.c)
if (WIN32)
    set(RYMGA_PLATFORM_SOURCE src/rymga_platform_windows.c)
endif()

add_library(rymga_loader_native STATIC src/rymga_loader.c ${RYMGA_PLATFORM_SOURCE})
target_include_directories(rymga_loader_native PUBLIC include PRIVATE ${SODIUM_INCLUDE_DIR})
target_link_libraries(rymga_loader_native PUBLIC CURL::libcurl PRIVATE ${SODIUM_LIBRARY})
if (RYMGA_SELF_CONTAINED AND WIN32)
    target_compile_definitions(rymga_loader_native PUBLIC CURL_STATICLIB=1)
endif()
if (WIN32)
    target_link_libraries(rymga_loader_native PRIVATE advapi32)
endif()
target_compile_features(rymga_loader_native PUBLIC c_std_11)
target_compile_options(rymga_loader_native PRIVATE -Wall -Wextra -Wpedantic -Werror)
if (RYMGA_ENABLE_FUZZER)
    target_compile_definitions(rymga_loader_native PRIVATE RYMGA_ENABLE_FUZZER=1)
    target_compile_options(rymga_loader_native PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
    add_executable(rymga_loader_manifest_fuzz tests/manifest_fuzz.c)
    target_link_libraries(rymga_loader_manifest_fuzz PRIVATE rymga_loader_native)
    target_compile_options(rymga_loader_manifest_fuzz PRIVATE -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
    target_link_options(rymga_loader_manifest_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()

set(RYMGA_LAUNCHER_SOURCE launcher/rymga_loader.c)
if (WIN32)
    set(RYMGA_LAUNCHER_SOURCE launcher/rymga_loader_windows.c)
endif()
add_executable(rymga-loader ${RYMGA_LAUNCHER_SOURCE})
target_include_directories(rymga-loader PRIVATE ${SODIUM_INCLUDE_DIR} src)
target_link_libraries(rymga-loader PRIVATE rymga_loader_native ${SODIUM_LIBRARY})
target_compile_options(rymga-loader PRIVATE -Wall -Wextra -Wpedantic -Werror)
if (WIN32 AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_link_options(rymga-loader PRIVATE -municode)
endif()

set_target_properties(rymga_loader_native PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(rymga_loader_jni SHARED jni/rymga_loader_jni.c)
target_include_directories(rymga_loader_jni PRIVATE ${JNI_INCLUDE_DIRS} ${SODIUM_INCLUDE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(rymga_loader_jni PRIVATE rymga_loader_native ${SODIUM_LIBRARY})
target_compile_options(rymga_loader_jni PRIVATE -Wall -Wextra -Wpedantic -Werror)
if (WIN32)
    set_target_properties(rymga_loader_jni PROPERTIES PREFIX "")
endif()

set(RYMGA_TEST_SOURCE tests/native_test.c)
if (WIN32)
    set(RYMGA_TEST_SOURCE tests/native_test_windows.c)
endif()
add_executable(rymga_loader_native_test ${RYMGA_TEST_SOURCE})
target_include_directories(rymga_loader_native_test PRIVATE src)
target_link_libraries(rymga_loader_native_test PRIVATE rymga_loader_native)
target_compile_options(rymga_loader_native_test PRIVATE -Wall -Wextra -Wpedantic -Werror)

enable_testing()
add_test(NAME rymga_loader_native_test COMMAND rymga_loader_native_test)

install(TARGETS rymga_loader_native rymga_loader_jni
        ARCHIVE DESTINATION lib
        LIBRARY DESTINATION lib
        RUNTIME DESTINATION bin)
install(TARGETS rymga-loader RUNTIME DESTINATION bin)
install(FILES include/rymga_loader.h DESTINATION include)
install(FILES ${CMAKE_CURRENT_SOURCE_DIR}/../../LICENSE DESTINATION share/licenses/rymga-low-level-loader-native)
if (RYMGA_SELF_CONTAINED)
    install(DIRECTORY ${RYMGA_THIRD_PARTY_LICENSE_DIR}/
            DESTINATION share/licenses/rymga-low-level-loader-native/third-party)
endif()

set(CPACK_PACKAGE_NAME "rymga-low-level-loader-native")
set(CPACK_PACKAGE_VERSION "0.1.0")
set(CPACK_PACKAGE_FILE_NAME "${CPACK_PACKAGE_NAME}-${CPACK_PACKAGE_VERSION}-${CMAKE_SYSTEM_NAME}")
set(CPACK_PACKAGE_DIRECTORY "${CMAKE_BINARY_DIR}/packages")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/../../LICENSE")
set(CPACK_GENERATOR "TGZ;ZIP")
include(CPack)
